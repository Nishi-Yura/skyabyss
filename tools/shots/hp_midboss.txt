# The four mid bosses before/after the high-poly pass (docs/design/15_high_poly.md section 8, assignment 2).
# ./gradlew runShots -Pscript=tools/shots/hp_midboss.txt -PshotsName=hp_midboss_before   (and _after)
# (camera y is the FEET position: the eye is 1.62 higher.)
# Output: run/shots/<shotsName>/shots/hp_midboss/*.png and shots.log (PERF line = ms per frame). NoAI mobs on a flat basalt floor, noon.
world shots flat
size 1280 720
timeout 600
render 8
clouds off

cmd time set noon
cmd weather clear
cmd gamemode spectator
cmd fill -30 -61 -30 30 -61 30 minecraft:smooth_basalt destroy
cmd kill @e[type=!minecraft:player,distance=..80]
hud off

# ---- whalebone captain (2.75 blocks tall, faces +z)
cmd summon skyabyss:whalebone_captain 0 -60 0 {NoAI:1b,Rotation:[0f,0f],PersistenceRequired:1b}

waitchunks
wait 60
shot captain_front
camera minecraft:overworld 3.4 -59.8 4.4 150 8
wait 20
shot captain_three_quarter
camera minecraft:overworld 0.3 -59.0 2.0 180 0
wait 20
shot captain_face
camera minecraft:overworld 0 -60.4 2.4 180 14
wait 20
shot captain_hands
cmd kill @e[type=skyabyss:whalebone_captain]
