# 2026-11-05: ティザー映像 (60 秒) の前半「物語」の素材。島の夜明け・鐘の都・岩鯨 (下から)・岩鯨の背。
# 録画は 1920x1080・30 fps。録るあいだはゲームの時計を 1/10 にして (tick rate 2)、絵を一枚も落とさない。
# 絵は run/shots/teaser/shots/teaser_a/<場面>/0000.png …。つなぐのは tools/gen/teaser/make_teaser.py
# bash tools/dev/mc_run.sh runShots --no-daemon -Pscript=tools/shots/teaser_a.txt -PshotsName=teaser
world teaser continent
size 1920 1080
timeout 12000
render 16
clouds on
guiscale 2
wait 100
cmd skyabyss cinematic stop
close
cmd skyabyss cinematic stop
cmd gamerule doDaylightCycle false
cmd gamerule doWeatherCycle false
cmd weather clear
cmd gamemode spectator
cmd skyabyss dev opening build
wait 200
hud off
chat clear

# A1 島の夜明け: 丘より高い所から、東の海と朝日へゆっくり進む
cmd time set 23300
cmd tp @s -1960 118 1872 -100 10
waitchunks 900
wait 80
chat clear
cmd tick rate 2
record a1_island 140 30
dolly 140 ~30 ~-6 ~-4 ~6 ~-2 linear
recwait
cmd tick rate 20

# A3 鐘の都: 南西の空から都へ寄る
cmd time set 23700
cmd skyabyss dev opening at shelter 0 -120 60 220 200 18
waitchunks 900
wait 60
chat clear
cmd tick rate 2
record a3_capital 140 30
dolly 140 ~40 ~-6 ~-40 ~-25 ~-4 linear
recwait
cmd tick rate 20

# A4 岩鯨: 下から見上げる。巨神は自分で泳ぐので、カメラはわずかに回るだけ
cmd time set 1500
cmd skyabyss progress grant @p stone_whale
lowpoly on
cmd tp @s 230 200 -880 180 0
waitchunks 600
cmd skyabyss titan summon
wait 200
cmd execute as @e[type=skyabyss:stone_whale,limit=1] at @s run tp @p ^-50 ^-30 ^95 facing entity @e[type=skyabyss:stone_whale,limit=1] feet
waitchunks 500
wait 60
chat clear
cmd tick rate 2
record a4_whale_low 180 30
dolly 180 ~6 ~-3 ~4 ~-8 ~3
recwait
cmd tick rate 20

# A5 岩鯨の背 (試練の島): 高い所から背の上を流す
cmd skyabyss dev arrive titan_back
waitchunks
wait 300
chat clear
hud off
camera skyabyss:titan_back -30 150 150 160 24
waitchunks 600
wait 60
chat clear
cmd tick rate 2
record a5_whale_back 140 30
dolly 140 ~40 ~-8 ~-20 ~25 ~4 linear
recwait
cmd tick rate 20
quit
