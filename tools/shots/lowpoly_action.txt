# Low-poly bone_sentinel in motion: walk, attack, hurt flash, night with a torch (docs/design/15_high_poly.md section 9).
# ./gradlew runShots -Pscript=tools/shots/lowpoly_action.txt
world shots flat
size 1280 720
timeout 600
render 8
clouds off

cmd time set noon
cmd weather clear
cmd gamemode spectator
cmd fill -20 -61 -20 20 -61 20 minecraft:smooth_basalt destroy
cmd kill @e[type=!minecraft:player,distance=..64]
hud off
camera minecraft:overworld 0 -60 0 180 0
perspective front
waitchunks
wait 20
cmd gamemode survival
cmd summon skyabyss:bone_sentinel 3.5 -60 0 {Rotation:[90f,0f],PersistenceRequired:1b}
wait 40
wait 5
shot lowpoly_move_00
wait 5
shot lowpoly_move_01
wait 5
shot lowpoly_move_02
wait 5
shot lowpoly_move_03
wait 5
shot lowpoly_move_04
wait 5
shot lowpoly_move_05
wait 5
shot lowpoly_move_06
wait 5
shot lowpoly_move_07
wait 5
shot lowpoly_move_08
wait 5
shot lowpoly_move_09
wait 5
shot lowpoly_move_10
wait 5
shot lowpoly_move_11
wait 5
shot lowpoly_move_12
wait 5
shot lowpoly_move_13
cmd damage @e[type=skyabyss:bone_sentinel,limit=1] 1 minecraft:generic
wait 1
shot lowpoly_hurt
cmd kill @e[type=skyabyss:bone_sentinel]
cmd gamemode spectator
perspective first
# night, torch light
cmd time set midnight
cmd setblock 1 -60 -4 minecraft:torch
cmd setblock -1 -60 -4 minecraft:torch
cmd summon skyabyss:bone_sentinel 0 -60 -4 {NoAI:1b,Rotation:[0f,0f],PersistenceRequired:1b}
camera minecraft:overworld 0 -59.5 1.3 180 0
wait 40
shot lowpoly_night_torch_front
camera minecraft:overworld 2.8 -59.5 -2 135 0
wait 10
shot lowpoly_night_torch_side
cmd kill @e[type=skyabyss:bone_sentinel]
cmd setblock 1 -60 -4 minecraft:air
cmd setblock -1 -60 -4 minecraft:air
quit
