# 祠の鎚 (Blockbench で手で組んだ模型) を撮る
# ./gradlew runShots --offline --no-daemon -Pscript=tools/shots/gear_dagger.txt -PshotsName=gear_dagger
world gear_look flat
size 1280 720
timeout 4000
render 6
clouds off
guiscale 2
cmd time set noon
cmd gamerule doWeatherCycle false
cmd weather clear
cmd gamemode survival
cmd gamerule doMobLoot false
camera 0 -60 0 0 0
hud off
cmd skyabyss cinematic stop
waitchunks
chat clear
wait 200
fov 70
cmd skyabyss level set @p 30
swingpose off
gearlook all a
cmd tp @s 0.5 -60 0.5 0 0
cmd kill @e[type=!player]
hud on
perspective first
fov 70
cmd tp @s 0.5 -60 0.5 0 0
cmd item replace entity @s weapon.mainhand with skyabyss:shrine_dagger
close
wait 30
chat clear
shot fp_shrine_dagger
hud off
perspective front
fov 30
cmd tp @s 0.5 -60 0.5 -60 0
wait 20
shot close_shrine_dagger
cmd tp @s 0.5 -60 0.5 -130 0
wait 20
shot side_shrine_dagger
fov 70
cmd tp @s 0.5 -60 0.5 -30 0
wait 20
shot full_shrine_dagger
fov 30
cmd tp @s 0.5 -60 0.5 120 0
wait 20
shot back_shrine_dagger
cmd tp @s 0.5 -60 0.5 -100 0
wait 20
shot right_shrine_dagger
fov 30
cmd tp @s 0.5 -60 0.5 -60 80
wait 20
shot pommel_shrine_dagger
