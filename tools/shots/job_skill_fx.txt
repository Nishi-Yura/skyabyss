# 職の新しい技の見た目 (静止画)。bash tools/dev/mc_run.sh runShots -Pscript=tools/shots/job_skill_fx.txt -PshotsName=job_skill_fx
world shots flat
size 1280 720
timeout 1500
render 6
clouds off
guiscale 2
cmd time set noon
cmd weather clear
cmd gamemode survival
cmd skyabyss level set @p 30
camera 0 -60 0 0 0
hud off
cmd skyabyss cinematic stop
waitchunks
chat clear
wait 200
perspective back
fov 60
# ---- thunder_bolt
cmd tp @s 0.5 -60 0.5 0 0
cmd skyabyss skill forget @s
cmd item replace entity @s weapon.mainhand with skyabyss:shrine_staff
wait 10
cmd skyabyss skill learn @s thunder_bolt 2
cmd skyabyss skill slot @s 1 thunder_bolt
cmd skyabyss skill cooldowns @s
cmd execute as @e[type=minecraft:husk] run tp @s ~ -400 ~
cmd summon minecraft:husk 0.5 -60 6.5 {NoAI:1b,Silent:1b,Health:500f,attributes:[{id:"minecraft:max_health",base:500.0d}]}
wait 10
cmd skyabyss skill mana @s full
cmd skyabyss skill cast @s 1
wait 3
shot fx_thunder_bolt
wait 30
