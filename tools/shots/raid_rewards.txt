# レイドの部屋の報い (docs/design/32 の 9.4): 部屋を越えたときの獲得の知らせ・試練の板の「称号」の行。
#   bash tools/dev/mc_run.sh runShots -Pscript=tools/shots/raid_rewards.txt -PshotsName=raid_rewards
#   絵は run/shots/raid_rewards/shots/raid_rewards/ → docs/review/previews/raid_rewards_*.png へ写す
world raid_rewards normal
size 1280 720
timeout 3000
render 12
clouds off

wait 100
cmd skyabyss cinematic stop
wait 10
cmd skyabyss cinematic stop
cmd skyabyss playtest skip opening
cmd time set noon
cmd weather clear
cmd gamerule doDaylightCycle false
cmd skyabyss playtest god on
cmd skyabyss playtest gear tide
# 四つの背の組み立てが終わるまで待つ (潮殻の背は組み立てに一分ほどかかる)
wait 1500
chat clear
# 歩の部屋 (潮殻) へ。先に潮殻の背へ入る (raid_tide と同じ段取り)、そのあと /skyabyss play raid が部屋へ運ぶ
cmd skyabyss playtest board tide
cmd skyabyss playtest goto tide hub
wait 100
waitchunks
wait 20
cmd skyroute enter
wait 30
waitchunks
waitband 600 第一の試練
wait 60
cmd skyabyss play raid tide walk
wait 1200
waitchunks
wait 40
hud on
# 部屋を越える (開発の段飛ばし) と、獲得の知らせが出る
cmd skyroute pass
wait 60
shot reward_notice_walk
wait 200
shot reward_notice_walk_late
chat clear

# 試練の板: 称号の行 (三つ渡してから開く。id は raid.<種類>.<巨神>)。GUI 2 と 3 で撮る
cmd skyabyss dev trial title raid.full.tide_shell
cmd skyabyss dev trial title raid.nowipe.tide_shell
cmd skyabyss dev trial title raid.rooms.tide_shell
guiscale 2
cmd skyabyss dev trial board tide_shell
wait 200
shot trial_board_titles_g2
guiscale 3
wait 40
shot trial_board_titles_g3
guiscale 4
wait 40
shot trial_board_titles_g4
wait 10
