# 章の跡地の小ダンジョン 6 つ (dungeon/MiniDungeons、docs/design/55 の 6 章): 部屋の層 (入口・部屋 A〜C・奥の光) と主の間
# bash tools/dev/mc_run.sh runShots -Pscript=tools/shots/mini_dungeons.txt -PshotsName=mini_dungeons
world mini_dungeons flat
size 1600 900
timeout 2400
render 10
guiscale 2
clouds off
wait 100
cmd skyabyss cinematic stop
close
cmd time set 6000
cmd weather clear
cmd gamerule doDaylightCycle false
cmd gamemode creative
hud off
chat clear
# ---- ch1_bone_pit
cmd skyabyss dungeon enter ch1_bone_pit
wait 100
cmd skyabyss cinematic stop
cmd skyabyss dungeon goto cp room_a
wait 40
shot ch1_bone_pit_1_entry
cmd skyabyss dungeon goto room_a room_b
wait 60
shot ch1_bone_pit_2_room_a

wait 80
