size 1280 720
timeout 1200
guiscale 3
wait 100
cmd skyabyss cinematic stop
close
cmd gamemode adventure
cmd skyabyss level set @p 30
cmd time set noon
cmd weather clear
cmd effect give @s resistance 600 4 true

wait 60
cmd skyabyss facility mount ride 60000
cmd item replace entity @p weapon.mainhand with skyabyss:mount_whistle

use

cmd execute if entity @e[type=skyabyss:mount] run say mount_at_10
cmd tp @s ~ ~ ~ 180 15
wait 2
shot mb_1_runs_to
wait 40
cmd execute if entity @e[type=skyabyss:mount] run say mount_at_52

wait 10
hud off
perspective back
wait 10
chat clear
shot mb_2_tassel
perspective front
wait 10
shot mb_3_tassel_front
