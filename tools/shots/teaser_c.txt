# 2026-11-05: ティザー映像 (60 秒) の後半「遊び」の素材 2。超大型レイド「あいだの塔」の主の技 (星屑の雨・淵の吐息・落ちる月)。
# 録画は 1920x1080・30 fps、録るあいだはゲームの時計を 1/10 (tick rate 2)
# bash tools/dev/mc_run.sh runShots --no-daemon -Pscript=tools/shots/teaser_c.txt -PshotsName=teaser
world teaser continent
size 1920 1080
timeout 9000
render 14
clouds on
guiscale 2
wait 100
cmd skyabyss cinematic stop
close
cmd skyabyss cinematic stop
cmd gamemode creative
hud off
chat clear
cmd skyabyss tower start
wait 60
cmd gamemode spectator
camera skyabyss:tower 0 127 36 180 -14
waitchunks 300
cmd skyabyss tower act hold
wait 460
cmd skyabyss cinematic stop
wait 10
chat clear

# C1 橋の上から主を見上げ、ゆっくり寄る
camera skyabyss:tower 0 124 40 180 -10
wait 20
cmd tick rate 2
record c1_tower_reveal 120 30
dolly 120 ~ ~6 ~-14 ~ ~-8
recwait
cmd tick rate 20

# C2 星屑の雨 (右手を空へ、星が橋へ落ちる)
camera skyabyss:tower 0 134 14 180 -22
wait 20
chat clear
cmd tick rate 2
record c2_tower_stars 110 30
wait 4
cmd skyabyss tower act stars
dolly 100 ~6 ~-2 ~6 ~-10 ~6 linear
recwait
cmd tick rate 20

# C3 淵の吐息 (縦の口から紫の炎) を横から
camera skyabyss:tower -60 140 -20 -110 2
wait 20
chat clear
cmd tick rate 2
record c3_tower_breath 110 30
wait 4
cmd skyabyss tower act breath
dolly 100 ~ ~ ~8 ~6 ~ linear
recwait
cmd tick rate 20

# C4 落ちる月が橋を進む
camera skyabyss:tower 0 150 60 180 -8
wait 20
chat clear
cmd tick rate 2
record c4_tower_moon 130 30
wait 4
cmd skyabyss tower act moon
dolly 120 ~ ~-6 ~-10 ~ ~-4
recwait
cmd tick rate 20
cmd skyabyss tower leave
wait 40
quit
