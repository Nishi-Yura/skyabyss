# 名前: <技>_1p_<a|b|c|d> (昼の一人称)、<技>_3p_<…> (昼の三人称)、<技>_night_1p_<b|c>、<技>_night_3p_<b|c>
# a = 溜め (撃った 1 tick 後)、b = 放ち、c = 当たり、d = 残り
world skill_fx flat
size 1280 720
timeout 3600
render 6
clouds off
guiscale 2
cmd time set noon
cmd weather clear
cmd gamemode survival
camera 0 -60 0 0 0
hud off
cmd skyabyss cinematic stop
waitchunks
chat clear
wait 200
