# 技の見た目 (撮影台本)。tools/gen/shots/gen_skill_fx_shots.py が作る。手で直さず、生成器を直す。
# 撮影: bash tools/dev/mc_run.sh runShots -Pscript=tools/shots/skill_fx.txt -PshotsName=skill_fx
# 一技 = 一つの動画 (record <技> <tick> <fps>)。run/shots/skill_fx/<技>/0000.png … に書く。
# 録画の間は /tick rate 5 (論理時刻で撮る)。終わったら tools/site/encode_skill_fx_clips.py で webp にする。
world skill_fx flat
size 1280 720
timeout 7200
render 6
clouds off
guiscale 2
cmd time set noon
cmd gamerule doWeatherCycle false
cmd weather clear
cmd gamemode survival
cmd gamerule doMobLoot false
camera 0 -60 0 0 0
hud off
cmd skyabyss cinematic stop
waitchunks
chat clear
wait 200
wait 240
cmd tick rate 5
wait 20

# ---- titan_cleave (昼・三人称の背中側。36 tick)
perspective back
fov 48
cmd tp @s 0.5 -60 0.5 0 0
cmd gamerule doWeatherCycle false
cmd weather clear
cmd skyabyss cinematic stop
huddemo off
hud off
wait 30
cmd skyabyss skill learn @s titan_cleave 3
cmd skyabyss skill slot @s 1 titan_cleave
cmd skyabyss skill cooldowns @s
cmd execute as @e[type=minecraft:husk] run tp @s ~ -400 ~
cmd execute if entity @e[type=minecraft:item] run kill @e[type=minecraft:item]
cmd summon minecraft:husk -1.0 -60 3.5 {NoAI:1b,Silent:1b,Health:500f,attributes:[{id:"minecraft:max_health",base:500.0d}]}
wait 10
cmd skyabyss skill mana @s full
record titan_cleave 36 30
wait 4
cmd skyabyss skill cast @s 1
recwait
cmd skyabyss skill cooldowns @s
wait 30
