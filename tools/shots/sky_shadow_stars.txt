# 空の影 (姿の層・うねり・雲の押しのけ) と、星の欠け (自前の星)。./gradlew runShots -Pscript=tools/shots/sky_shadow_stars.txt
world shots flat
size 1280 720
timeout 600
render 12
clouds on

cmd gamemode creative
cmd time set noon
cmd weather clear
hud off
waitchunks
cmd skyabyss opening reset
chat clear
wait 5
cmd skyabyss opening shadow
camera 0 80 0 0 -55
wait 330
shot shadow_a
wait 120
shot shadow_b
wait 120
look 0 -75

wait 100
shot shadow_d
wait 150
