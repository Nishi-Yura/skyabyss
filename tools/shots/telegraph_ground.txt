# Floor decals drawn only on walkable floor (docs/design/02_stone_whale.md section 5.4, 'floor decal')
# ./gradlew runShots -Pscript=tools/shots/telegraph_ground.txt   (mute: the shots runner sets the volume to 0)
# Five scenes (flat, steps+stairs, wall+pillars, mobs, low ceilings) around P = (0.5, 12.5). The captain's stomp circle (radius 4) forms under the player,
# then the player turns to spectator and the camera looks from an oblique height (obl_*) and from the floor (eye_*).
world telegraph_ground flat
size 1280 720
timeout 900
render 8
clouds off

cmd time set noon
cmd weather clear
cmd gamerule doDaylightCycle false
cmd gamerule doWeatherCycle false
cmd gamemode survival Dev
cmd effect give Dev minecraft:resistance 1000000 4 true
cmd fill -30 -61 -30 30 -61 30 minecraft:smooth_basalt destroy
hud off
camera minecraft:overworld 0.5 -60 12.5 180 0
waitchunks
wait 20
cmd skyabyss cinematic stop
chat clear

# ---------------- scene: flat
cmd time set noon
cmd weather clear
cmd kill @e[type=!minecraft:player,distance=..80]
cmd gamemode survival Dev
camera minecraft:overworld 0.5 -60 12.5 180 0
cmd fill -8 -60 5 10 -46 24 minecraft:air
wait 10
cmd kill @e[type=skyabyss:whalebone_captain]
cmd gamemode survival Dev
camera minecraft:overworld 0.5 -60 12.5 180 0
cmd summon skyabyss:whalebone_captain 0 -60 0 {PersistenceRequired:1b,Rotation:[0f,0f]}
wait 25
cmd skyabyss dev force @e[type=skyabyss:whalebone_captain,limit=1] LEAP
wait 1
cmd gamemode spectator Dev
camera minecraft:overworld 0.5 -52 22.5 180 38
wait 3
shot obl_flat
wait 6
shot obl_flat_b
wait 30
cmd kill @e[type=skyabyss:whalebone_captain]
cmd gamemode survival Dev
camera minecraft:overworld 0.5 -60 12.5 180 0
cmd summon skyabyss:whalebone_captain 0 -60 0 {PersistenceRequired:1b,Rotation:[0f,0f]}
wait 25
cmd skyabyss dev force @e[type=skyabyss:whalebone_captain,limit=1] LEAP
wait 1
cmd gamemode spectator Dev
camera minecraft:overworld 0.5 -59 21.5 180 6
wait 3
shot eye_flat
wait 6
shot eye_flat_b
wait 30
