# The ash smith's telegraphed attacks for a party of 6 (docs/design/13_ember.md section 18)
# ./gradlew runShots -Pscript=tools/shots/telegraph_ember_smith.txt   (mute: the shots runner sets the volume to 0)
# One player on a barrier platform 20 blocks above the floor; "dev emberforce <smith> <attack> 6" acts as if 6 players were in the room
# (the nearby player is made the smith's fighter, so the smith works outside its room).
world shots flat
size 1280 720
timeout 900
render 8
clouds off

cmd time set noon
cmd weather clear
cmd gamemode survival Dev
cmd effect give Dev minecraft:resistance 1000000 4 true
cmd fill -30 -61 -30 30 -61 30 minecraft:smooth_basalt destroy
cmd fill -4 -41 19 4 -41 25 minecraft:air
cmd fill -4 -41 19 4 -41 25 minecraft:barrier
cmd kill @e[type=!minecraft:player,distance=..80]
hud off

camera minecraft:overworld 0.5 -40 22.5 180 40
waitchunks
wait 20
cmd skyabyss cinematic stop
chat clear
cmd summon skyabyss:ash_smith 0 -60 0 {PersistenceRequired:1b,Rotation:[0f,0f]}
wait 10
cmd skyabyss cinematic stop
camera minecraft:overworld 0.5 -40 22.5 180 40
wait 60

camera minecraft:overworld 0.5 -40 22.5 180 40
wait 5
cmd skyabyss dev emberforce @e[type=skyabyss:ash_smith,limit=1] SLAM 6
wait 10
shot slam_a
wait 12
shot slam_hit
wait 20
shot slam_vent
wait 50

camera minecraft:overworld 0.5 -40 22.5 180 40
wait 5
cmd skyabyss dev emberforce @e[type=skyabyss:ash_smith,limit=1] COLD_SLAM 6
wait 12
shot cold_a
wait 14
shot cold_b
wait 16
shot cold_c
wait 60

camera minecraft:overworld 0.5 -40 22.5 180 40
wait 5
cmd skyabyss dev emberforce @e[type=skyabyss:ash_smith,limit=1] BREATH 6
wait 14
shot breath_a
wait 24
shot breath_hit
wait 60

camera minecraft:overworld 0.5 -40 22.5 180 40
wait 5
cmd skyabyss dev emberforce @e[type=skyabyss:ash_smith,limit=1] SPARKS 6
wait 26
shot sparks_a
wait 14
shot sparks_b
wait 60
cmd kill @e[type=skyabyss:ash_smith]
quit
