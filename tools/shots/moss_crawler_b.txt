# ./gradlew runShots -Pscript=tools/shots/moss_crawler_b.txt   (出力 docs/review/previews/moss_b_*.png へ手でコピー)
world shots flat
size 1280 720
timeout 600
render 8
clouds off

cmd skyabyss cinematic stop
cmd time set noon
cmd weather clear
cmd gamemode spectator
cmd fill -20 -61 -20 20 -61 20 minecraft:smooth_basalt destroy
cmd kill @e[type=!minecraft:player,distance=..64]
cmd summon skyabyss:moss_crawler 0 -60 0 {NoAI:1b,Rotation:[0f,0f],PersistenceRequired:1b,Invulnerable:1b,Tags:["mc"]}
hud off
camera minecraft:overworld 0 -61.0 5.5 180 4
waitchunks
wait 40
cmd skyabyss cinematic stop
wait 40
fov 40
shot moss_b_front
camera minecraft:overworld -5.5 -61.2 0 270 4
wait 15
shot moss_b_side
camera minecraft:overworld 0 -61.0 -5.5 0 4
wait 15
shot moss_b_back
camera minecraft:overworld 0 -57.1 0.2 180 90
wait 15
shot moss_b_top

wait 15
shot moss_b_face
