# 体ごとのスキンを一人ずつ四つの向きで。空に足場を置いて撮る (地面の高さで位置がずれないように)
world continent_areas continent
size 1280 720
timeout 1200
render 6
guiscale 3
wait 100
cmd skyabyss cinematic stop
close
cmd gamemode adventure
cmd time set 6000
cmd weather clear

wait 60
cmd skyabyss level set @p 50
cmd item replace entity @s armor.head with minecraft:air
cmd item replace entity @s armor.chest with minecraft:air
cmd item replace entity @s armor.legs with minecraft:air
cmd item replace entity @s armor.feet with minecraft:air
cmd skyabyss skin give all
cmd skyabyss skin wear witch
hud off
chat clear
cmd fill -126 159 -606 -114 159 -592 minecraft:smooth_stone
