# bash tools/dev/mc_run.sh runShots -Pscript=tools/shots/skill_props.txt -PshotsName=skill_props

size 1280 720
timeout 2400
render 6
clouds off
guiscale 2
cmd time set noon
cmd gamerule doWeatherCycle false
cmd weather clear
cmd gamemode survival
cmd gamerule doMobLoot false

hud off
cmd skyabyss cinematic stop
waitchunks
chat clear
wait 200
perspective back
fov 60
cmd skyabyss level set @p 30
