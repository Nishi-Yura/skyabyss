# 番人が倒れたあとの場面 (abyss_warden_fallen) を撮る
# bash tools/dev/mc_run.sh runShots -Pscript=tools/shots/abyss_review.txt -PshotsName=abyss_review
world abyss_fallen flat
size 1280 720
timeout 5400
render 10
clouds off

wait 80
cmd skyabyss cinematic stop
cmd gamemode creative
cmd effect give @s night_vision 99999 0 true

cmd skyabyss abyss enter
wait 40
cmd skyabyss abyss build
wait 1200
cmd skyabyss abyss layer go heart 180
wait 5
cmd skyabyss cinematic stop
cmd tp @s 0 -298 12 180 5
wait 20
hud on
cmd skyabyss cinematic play abyss_warden_fallen
wait 20
shot w1
wait 60
shot w2
wait 60
shot w3
wait 60
shot w4
