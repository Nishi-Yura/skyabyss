# Low-poly mesh pipeline check on bone_sentinel (docs/design/15_high_poly.md section 9).
# ./gradlew runShots -Pscript=tools/shots/lowpoly_basic.txt
world shots flat
size 1280 720
timeout 600
render 8
clouds off

cmd time set noon
cmd weather clear
cmd gamemode spectator
cmd fill -20 -61 -20 20 -61 20 minecraft:smooth_basalt destroy
cmd kill @e[type=!minecraft:player,distance=..64]
cmd summon skyabyss:bone_sentinel 0 -60 0 {NoAI:1b,Rotation:[0f,0f],PersistenceRequired:1b}
hud off
camera minecraft:overworld 0 -59.5 5.2 180 0
waitchunks
wait 60
shot lowpoly_idle_front
lowpoly debug
wait 5
shot lowpoly_align_front
camera minecraft:overworld 5.2 -59.5 0 90 0
wait 10
shot lowpoly_align_side
lowpoly nodebug
wait 5
shot lowpoly_idle_side
camera minecraft:overworld 0 -59.5 -5.2 0 0
wait 10
shot lowpoly_idle_back
lowpoly off
wait 5
camera minecraft:overworld 0 -59.5 5.2 180 0
wait 10
shot cubes_idle_front
quit
