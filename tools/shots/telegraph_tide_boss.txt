# The tide beckoner (mid-boss)'s telegraphed patterns for a party of 6 (docs/design/08_tide_shell.md section 5.2 / 5.5)
# ./gradlew runShots -Pscript=tools/shots/telegraph_tide_boss.txt   (mute: the shots runner sets the volume to 0)
# One player on a barrier platform 17 blocks above the floor; "dev tideforce <boss> <attack> 6" acts as if 6 players were fighting.
world shots flat
size 1280 720
timeout 900
render 8
clouds off

cmd time set noon
cmd weather clear
cmd gamemode survival Dev
cmd effect give Dev minecraft:resistance 1000000 4 true
cmd fill -30 -61 -30 30 -61 30 minecraft:sand destroy
cmd fill -4 -44 11 4 -44 17 minecraft:air
cmd fill -4 -44 11 4 -44 17 minecraft:barrier
cmd kill @e[type=!minecraft:player,distance=..80]
hud off

waitchunks
wait 60
camera minecraft:overworld 0.5 -43 14.5 180 38
waitchunks
wait 20
cmd skyabyss cinematic stop
chat clear
cmd summon skyabyss:tide_beckoner 0 -60 0 {PersistenceRequired:1b,Rotation:[0f,0f]}
wait 10
cmd skyabyss cinematic stop
camera minecraft:overworld 0.5 -43 14.5 180 38
wait 40

cmd skyabyss dev tideforce @e[type=skyabyss:tide_beckoner,limit=1] SLAM 6
wait 12
shot slam_a
wait 70

cmd skyabyss dev tideforce @e[type=skyabyss:tide_beckoner,limit=1] BREATH 6
wait 14
shot breath_a
wait 40

cmd skyabyss dev tideforce @e[type=skyabyss:tide_beckoner,limit=1] BURROW 6
wait 36
shot burrow_a
wait 30

cmd skyabyss dev tideforce @e[type=skyabyss:tide_beckoner,limit=1] ROAR 6
wait 16
shot roar_a
wait 18
shot roar_b
wait 40

cmd kill @e[type=skyabyss:tide_beckoner]
quit
