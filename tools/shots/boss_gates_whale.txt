# 岩鯨の背のボス部屋の大門 (心臓の間・番兵長の広間) を撮る。./gradlew runShots -Pscript=tools/shots/boss_gates_whale.txt
# 出力: run/shots/boss_gates_whale/shots/boss_gates_whale/*.png → python tools/boss_gates_sheet.py で docs/review/previews/boss_gates_sheet.png
# 四つの背を建て終えた専用サーバーへ入る (標準の平らな世界だと 4 つの背を一度に建てて、読み込みが間に合わない)。
# サーバー: run/server.properties の server-port=25726、先に OP を渡す: python tools/rcon.py --port 25826 "op Shooter"。起動: -PshotsUser=Shooter
server localhost:25726
size 1280 720
timeout 900
render 16
clouds off

cmd time set midnight
cmd weather clear
cmd gamemode spectator
hud off
fov 85
# 心臓の間: 回廊 (休み場の北) から大門を見る (北向き。yaw 180)
camera skyabyss:titan_back 0.5 95.6 -49.5 180 -12
waitchunks 900
wait 60
shot gate_whale_heart
# 番兵長の広間: 控えの間から、扉の先の大門を見る (西向き。yaw 90)
camera skyabyss:titan_back 31.5 95.6 -38.5 90 -12
waitchunks 900
wait 30
shot gate_whale_sentinel
quit
