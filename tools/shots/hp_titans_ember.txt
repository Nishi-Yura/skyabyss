# Ember titan (near LOD high-poly) before / after + frame time near the body.
# ./gradlew runShots -Pscript=tools/shots/hp_titans_ember.txt -PshotsName=hp_titans_ember_before   (and _after)
# Coordinates are local to the titan (^ = left ^ up ^ forward; the head is forward, about 103 blocks from the centre). No OS input.
world hp_titans_ember flat
size 1280 720
timeout 900
render 16
clouds off

cmd time set 18000
cmd weather clear
cmd gamemode creative
cmd skyabyss progress grant @p thunder_wing
camera 0.5 -60 0.5 0 0
waitchunks
cmd skyabyss titan summon ember
wait 60
cmd skyabyss titan rest ember here
wait 60
hud off
cmd gamemode spectator
cmd execute as @e[type=skyabyss:ember_titan,limit=1] at @s run tp @p ^-14 ^20 ^128 facing ^0 ^8 ^103
waitchunks
wait 40
shot head_front
cmd execute as @e[type=skyabyss:ember_titan,limit=1] at @s run tp @p ^50 ^26 ^118 facing ^0 ^8 ^96
wait 30
shot head_three_quarter

wait 30
shot head_top
# the ridge and the lava cracks

wait 30
shot ridge_close

wait 30
shot near_side
# the front left foot
cmd execute as @e[type=skyabyss:ember_titan,limit=1] at @s run tp @p ^90 ^8 ^-40 facing ^62 ^3 ^-44
