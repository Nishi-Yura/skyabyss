# 余燼の背のボス部屋の大門 (灰の鍛え手の扉・熾の底の入口) を撮る。./gradlew runShots -Pscript=tools/shots/boss_gates_ember.txt
# 出力: run/shots/boss_gates_ember/shots/boss_gates_ember/*.png → python tools/boss_gates_sheet.py で docs/review/previews/boss_gates_sheet.png
# 四つの背を建て終えた専用サーバーへ入る (標準の平らな世界だと 4 つの背を一度に建てて、読み込みが間に合わない)。
# サーバー: run/server.properties の server-port=25726、先に OP を渡す: python tools/rcon.py --port 25826 "op Shooter"。起動: -PshotsUser=Shooter
server localhost:25726
size 1280 720
timeout 900
render 16
clouds off

cmd time set midnight
cmd weather clear
cmd gamemode spectator
hud off
fov 85
# 鍛え手の扉: 鍛えの間の入口の前室から、大炉の向こうの大門を見る (南向き。yaw 0)
camera skyabyss:ember_back 26.5 83.0 -2.5 0 -8
waitchunks 900
wait 60
shot gate_ember_smith
# 熾の底: 前室の北の端 (螺旋の底からの通路) から大門を見る (南向き)
camera skyabyss:ember_back 0.5 52.6 -31.5 0 -25
waitchunks 900
wait 30
shot gate_ember_heart
quit
