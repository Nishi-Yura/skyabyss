# The captain's stomp and sweep floor decals seen from a normal player eye height (docs/design/02_stone_whale.md section 5.4)
# ./gradlew runShots -Pscript=tools/shots/telegraph_stomp_eye.txt   (mute: the shots runner sets the volume to 0)
# The player stands on the floor (survival + resistance V, so the hits do nothing); the stomp circle forms around the player, the sweep cone in front of the captain.
world shots flat
size 1280 720
timeout 600
render 8
clouds off

cmd time set noon
cmd weather clear
cmd gamemode survival Dev
cmd effect give Dev minecraft:resistance 1000000 4 true
cmd fill -30 -61 -30 30 -61 30 minecraft:smooth_basalt destroy
cmd kill @e[type=!minecraft:player,distance=..80]
hud off

# ---- stomp: standing at the edge of where the circle will land, looking down at it
cmd summon skyabyss:whalebone_captain 0 -60 10 {PersistenceRequired:1b,Rotation:[180f,0f]}
camera minecraft:overworld 0.5 -60 -2.5 0 32
waitchunks
wait 20
cmd skyabyss cinematic stop
chat clear
wait 40
cmd skyabyss dev force @e[type=skyabyss:whalebone_captain,limit=1] LEAP
wait 6
shot eye_stomp_a
wait 8
shot eye_stomp_b
wait 8
shot eye_stomp_c
wait 5
shot eye_stomp_hit
wait 40
cmd kill @e[type=skyabyss:whalebone_captain]

# ---- sweep: the cone in front of the captain, seen from beside it at eye height
cmd summon skyabyss:whalebone_captain 0 -60 0 {PersistenceRequired:1b,Rotation:[0f,0f]}
camera minecraft:overworld 0.5 -60 9.5 180 22
wait 40
cmd skyabyss dev force @e[type=skyabyss:whalebone_captain,limit=1] SWEEP
wait 6
shot eye_sweep_a
wait 10
shot eye_sweep_b
wait 6
shot eye_sweep_hit
cmd kill @e[type=skyabyss:whalebone_captain]
quit
