# 公式サイトの足りない絵 (site_showcase.txt・site_titans.txt の続き): 土地の町、都の三人称、巨神の背の順路の部屋、五章の深淵
# 町は data/skyabyss/continent/layout.json の towns (鯨見 -120,-760 / 汐守 1330,160 / 風待 -1180,-100 / 灰炉 -300,1100)
# bash tools/dev/mc_run.sh runShots -Pscript=tools/shots/site_extra.txt -PshotsName=site_showcase
world site_showcase continent
size 1920 1080
timeout 3600
render 16
clouds off
guiscale 2
wait 100
cmd skyabyss cinematic stop
close
cmd skyabyss cinematic stop
cmd time set 3000
cmd weather clear
cmd gamerule doDaylightCycle false
cmd gamerule doWeatherCycle false
cmd gamemode spectator
hud off
chat clear
lowpoly on

# ---- 土地の町 (鯨見・汐守)
cmd tp @s -30 135 -680 0 0
cmd execute as @s run tp @s -30 135 -680 facing -120 72 -760
waitchunks 600
wait 60
shot x01_town_whale
cmd tp @s 1240 130 240 0 0
cmd execute as @s run tp @s 1240 130 240 facing 1330 70 160
waitchunks 600
wait 60
shot x02_town_tide
cmd tp @s -1090 150 -20 0 0
cmd execute as @s run tp @s -1090 150 -20 facing -1180 90 -100
waitchunks 600
wait 60
shot x03_town_storm

# ---- 都の三人称 (目的の帯・体力・技の枠が見える)
