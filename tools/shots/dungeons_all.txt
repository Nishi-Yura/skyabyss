# 地下のダンジョン四つ (docs/design/55): 大陸の入口 (岩の口と入口の間)・各層の見どころ・主
# bash tools/dev/mc_run.sh runShots -Pscript=tools/shots/dungeons_all.txt -PshotsName=dungeons_all
world dungeons_all continent
size 1600 900
timeout 7200
render 10
guiscale 2
clouds off
wait 100
cmd skyabyss cinematic stop
close
cmd gamerule doDaylightCycle false
cmd gamerule doWeatherCycle false
cmd skyabyss dev opening build
wait 200
cmd gamemode creative
hud off
chat clear
# ---- bone_pit
cmd gamemode creative
cmd tp @s 60 190 -768 180 30
waitchunks 600
wait 900
cmd time set 6000
cmd weather clear
cmd skyabyss cinematic stop
cmd skyabyss dungeon mouth bone_pit
waitchunks 300
wait 80
shot bone_pit_mouth
cmd skyabyss dungeon hall bone_pit
waitchunks 200
wait 60
shot bone_pit_hall_gate
cmd execute as @s at @s run tp @s ~ ~ ~ ~180 5
wait 20
shot bone_pit_hall_back
cmd skyabyss dungeon enter bone_pit
wait 80
cmd gamemode spectator
shot bone_pit_f1_entry
cmd skyabyss dungeon goto view_dig view_dig_to
wait 50
shot bone_pit_dig
cmd skyabyss dungeon goto view_hall view_hall_to
wait 50
shot bone_pit_hall
cmd skyabyss dungeon goto view_cart view_cart_to
wait 50
shot bone_pit_cart
cmd skyabyss dungeon floor 1
wait 80
cmd skyabyss dungeon goto view_rim view_rim_to
wait 50
shot bone_pit_rim
cmd skyabyss dungeon goto view_bottom view_bottom_to
wait 50
shot bone_pit_bottom
cmd gamemode creative
cmd skyabyss dungeon floor 2
wait 60
cmd skyabyss dungeon goto view_lair view_lair_to
wait 160
cmd gamemode spectator
wait 20
shot bone_pit_lair
cmd skyabyss dungeon boss
wait 120
cmd skyabyss dungeon goto view_lair exit
wait 40
shot bone_pit_exit
cmd skyabyss dungeon leave
wait 40
