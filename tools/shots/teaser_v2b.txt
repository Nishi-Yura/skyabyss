# 2026-11-05: ティザー映像 二回目の「遊び」2。岩鯨の背を駆ける・地下のダンジョンの主と戦う (横から)・あいだの塔の主の技を避ける。
# 録画は 1920x1080・30 fps、録るあいだはゲームの時計を 1/10 (tick rate 2)
# bash tools/dev/mc_run.sh runShots --no-daemon -Pscript=tools/shots/teaser_v2b.txt -PshotsName=teaser
world teaser continent
size 1920 1080
timeout 14000
quality 24 12
guiscale 2
wait 100
cmd skyabyss cinematic stop
close
cmd skyabyss cinematic stop
cmd gamerule doDaylightCycle false
cmd gamerule doWeatherCycle false
cmd weather clear
cmd skyabyss level set @p 50
cmd skyabyss job choose warrior
cmd item replace entity @s weapon.mainhand with skyabyss:divine_sky_sword
cmd item replace entity @s armor.head with skyabyss:divine_bellkeeper_helmet
cmd item replace entity @s armor.chest with skyabyss:divine_bellkeeper_chestplate
cmd item replace entity @s armor.legs with skyabyss:divine_bellkeeper_leggings
cmd item replace entity @s armor.feet with skyabyss:divine_bellkeeper_boots
cmd skyabyss skill learn @s titan_cleave 3
cmd skyabyss skill learn @s whale_sweep 3
cmd skyabyss skill slot @s 1 titan_cleave
cmd skyabyss skill slot @s 2 whale_sweep
hud off
chat clear

# ---------------- W1 岩鯨の背を駆ける (右後ろから追う)
cmd gamemode spectator
cmd time set 11000
cmd skyabyss dev arrive titan_back
waitchunks
wait 300
camera skyabyss:titan_back 0 150 140 180 0
waitchunks 900
cmd gamemode adventure
cmd effect give @s resistance 99999 4 true
cmd effect give @s slow_falling 6 0 true
wait 160
chat clear
cam player 2.5 1.2 -5
camfollow 2.5 1.2 -5 0.2
wait 10
cmd tick rate 2
record w1_back_run 110 30
hold forward on
hold sprint on
wait 30
hold jump on
wait 5
hold jump off
wait 40
aim 20 ~25 ~
recwait
hold forward off
hold sprint off
cmd tick rate 20
camoff

wait 10
cmd skyabyss skill mana @s full

attack

cmd execute as @s at @s facing entity @e[type=!minecraft:player,distance=..30,limit=1,sort=nearest] eyes run tp @s ~ ~ ~ ~ ~

cmd skyabyss skill cast @s 2
