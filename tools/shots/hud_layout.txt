# HUD の置き場の確認 (docs/design/18_notices.md の 9 章)。./gradlew runShots -Pscript=tools/shots/hud_layout.txt [-PshotsName=<名前>]
# 一番混む場面 3 つを、GUI 倍率 2・3・4 で撮る:
#   boss    : パーティでのボス戦 (ボスバー 3 本・効き目・技の名前・手引き・キーの手がかり・目的の帯・誓い・目盛りと目的の板・仲間の一覧と矢印)
#   opening : 序盤 (手引き 2 行 + できた + キーの手がかり + 字幕 + 目的の帯)
#   back    : 背中 (目盛りと目的の板 + 誓い + 仲間の一覧 + 札)
# 見本のデータは huddemo (client/hud/HudDemo)。4:3 の窓は `size 1024 768` に替えて流す
world hud_layout flat
size 1280 720
timeout 900
render 8
clouds off

cmd time set noon
cmd weather clear
cmd gamemode survival
hud on
waitchunks
chat clear
guiscale 2
cmd effect give @s minecraft:speed infinite 0 true
cmd effect give @s minecraft:strength infinite 0 true
cmd effect give @s minecraft:night_vision infinite 0 true
cmd effect give @s minecraft:weakness infinite 0 true
cmd bossbar add hud:b1 "雷翼の心臓"
cmd bossbar set hud:b1 players @a
cmd bossbar set hud:b1 value 70
cmd bossbar set hud:b1 color red
cmd bossbar add hud:b2 "砦の番兵"
cmd bossbar set hud:b2 players @a
cmd bossbar set hud:b2 value 40
cmd bossbar add hud:b3 "雷獣"
cmd bossbar set hud:b3 players @a
cmd bossbar set hud:b3 value 90
cmd bossbar set hud:b3 color purple
wait 20

echo === boss fight in a party
huddemo boss
wait 75
shot boss_s2
guiscale 3
wait 30
shot boss_s3
guiscale 4
wait 30
shot boss_s4
huddemo off
cmd bossbar remove hud:b1
cmd bossbar remove hud:b2
cmd bossbar remove hud:b3
guiscale 2
chat clear
wait 20

echo === opening with tutorial and key hints
huddemo opening
wait 60
shot opening_s2
guiscale 3
wait 30
shot opening_s3
guiscale 4
wait 30
shot opening_s4
huddemo off
guiscale 2
chat clear
wait 20

echo === back with goal panel, vow and party
huddemo back
wait 60
shot back_s2
guiscale 3
wait 30
shot back_s3
guiscale 4
wait 30
shot back_s4
huddemo off
echo done
