# 仕立て屋 (スキンの店) が都と町の武具屋のそばに立つか、話すと今週の仕立てが開くか
# bash tools/dev/mc_run.sh runShots -Pscript=tools/shots/tailor.txt -PshotsName=tailor
world box_openers continent
size 1280 720
timeout 9000
render 10
guiscale 2
clouds off
waitchunks 300
wait 40
cmd time set 3000
cmd weather clear
cmd gamerule doDaylightCycle false
cmd skyabyss cinematic stop
close
cmd skyabyss dev opening build
wait 200
cmd gamemode creative
hud off
cmd skyabyss dev opening at shelter 0 0 2 20 180 6
waitchunks 600
wait 300
cmd execute at @e[type=skyabyss:attendant,nbt={Role:"outfitter"},limit=1,sort=nearest] run tp @s ~ ~1 ~
waitchunks 300
wait 300
cmd gamemode spectator
cmd execute at @e[type=skyabyss:attendant,nbt={Role:"tailor"},limit=1,sort=nearest] run tp @s ^ ^1.6 ^2.6 facing ~ ~1.4 ~
wait 8
chat clear
shot capital_tailor
cmd execute at @e[type=skyabyss:attendant,nbt={Role:"tailor"},limit=1,sort=nearest] run tp @s ^1.5 ^3.2 ^5.0 facing ~ ~1 ~
wait 8
shot capital_tailor_wide
cmd gamemode creative
hud on
cmd skyabyss dev talk "skyabyss:attendant_tailor"
wait 160
shot tailor_talk
