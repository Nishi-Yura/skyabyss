# Per-model render cost of the mid bosses: 6 copies of one model in view, 300 ticks each (docs/design/15_high_poly.md section 8).
# ./gradlew runShots -Pscript=tools/shots/hp_midboss_perf.txt -PshotsName=hp_midboss_perf_before   (and _after). PERF lines in shots.log.
world shots flat
size 1280 720
timeout 600
render 8
clouds off

cmd time set noon
cmd weather clear
cmd gamemode spectator
cmd fill -30 -61 -30 30 -61 30 minecraft:smooth_basalt destroy
cmd kill @e[type=!minecraft:player,distance=..80]

camera minecraft:overworld 0 -55 16 180 16
waitchunks
wait 40
