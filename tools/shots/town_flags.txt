# 土地の旗: 旗を持つ組合の旗を、土地の町の門と広場に掛ける (endgame/TownFlags。docs/design/54 の 5.1.1)
# bash tools/dev/mc_run.sh runShots -Pscript=tools/shots/town_flags.txt -PshotsName=town_flags
world continent_towns2 continent
size 1600 900
timeout 1800
render 12
clouds off
guiscale 2
wait 100
cmd skyabyss cinematic stop
wait 10
cmd time set 5000
cmd weather clear
cmd gamerule doDaylightCycle false
cmd skyabyss dev opening build
wait 200
cmd gamemode creative
cmd guild disband confirm
cmd guild create "Kujira" KJR blue yellow
cmd guild pattern whale
cmd skyabyss townflag respot whale
cmd skyabyss townflag set whale
hud off
cmd skyabyss dev grounds mark town 0 pos
waitchunks 900
wait 260
cmd execute at @s positioned ~ ~ ~-74 positioned over motion_blocking_no_leaves run tp @s ~ ~5 ~ 0 8
wait 80
shot town_flags_1_gate_out
cmd skyabyss dev grounds mark town 0 pos
cmd execute at @s positioned ~ ~ ~-34 positioned over motion_blocking_no_leaves run tp @s ~ ~4 ~ 180 6
wait 60
shot town_flags_2_gate_in
cmd skyabyss townflag goto whale
wait 60
shot town_flags_3_plaza
cmd skyabyss townflag clear whale
wait 140
shot town_flags_4_plaza_cleared
cmd skyabyss dev grounds mark town 0 pos
cmd execute at @s positioned ~ ~ ~-74 positioned over motion_blocking_no_leaves run tp @s ~ ~5 ~ 0 8
wait 140
shot town_flags_5_gate_cleared
