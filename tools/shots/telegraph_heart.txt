# The stone whale heart's telegraphed patterns for a party of 6 (docs/design/02_stone_whale.md section 5.3 / 5.4)
# ./gradlew runShots -Pscript=tools/shots/telegraph_heart.txt   (mute: the shots runner sets the volume to 0)
# One player on a barrier platform 17 blocks above the floor; "force <heart> <attack> 6" acts as if 6 players were in the room.
world shots flat
size 1280 720
timeout 900
render 8
clouds off

cmd time set noon
cmd weather clear
cmd gamemode survival Dev
cmd effect give Dev minecraft:resistance 1000000 4 true
cmd fill -30 -61 -30 30 -61 30 minecraft:smooth_basalt destroy
cmd fill -4 -44 11 4 -44 17 minecraft:air
cmd fill -4 -44 11 4 -44 17 minecraft:barrier
cmd kill @e[type=!minecraft:player,distance=..80]
hud off

camera minecraft:overworld 0.5 -43 14.5 180 38
waitchunks
wait 20
cmd skyabyss cinematic stop
chat clear
cmd summon skyabyss:stone_whale_heart 0 -59 0 {PersistenceRequired:1b,Rotation:[0f,0f]}
wait 10
cmd skyabyss cinematic stop
camera minecraft:overworld 0.5 -43 14.5 180 38
wait 100

cmd skyabyss dev force @e[type=skyabyss:stone_whale_heart,limit=1] THRUST 6
wait 12
shot thrust_a
wait 50

cmd skyabyss dev force @e[type=skyabyss:stone_whale_heart,limit=1] LEAP 6
wait 14
shot leap_a
wait 22
shot leap_b
wait 60

cmd skyabyss dev force @e[type=skyabyss:stone_whale_heart,limit=1] CHAIN 6
wait 16
shot chain_a
wait 22
shot chain_b
wait 70

cmd skyabyss dev force @e[type=skyabyss:stone_whale_heart,limit=1] BEAM 6
wait 24
shot beam_a
wait 26
shot beam_b
wait 60

cmd skyabyss dev force @e[type=skyabyss:stone_whale_heart,limit=1] SHELL_DROP 6
wait 18
shot drop_a
wait 60

cmd skyabyss dev force @e[type=skyabyss:stone_whale_heart,limit=1] CRACK 6
wait 18
shot crack_a
wait 60

cmd skyabyss dev force @e[type=skyabyss:stone_whale_heart,limit=1] VOLLEY 6
wait 18
shot volley_a
wait 60

cmd skyabyss dev force @e[type=skyabyss:stone_whale_heart,limit=1] SWEEP 6
wait 40
shot sweep_a
wait 20
shot sweep_b
cmd kill @e[type=skyabyss:stone_whale_heart]
quit
