# 余燼の三つの試練 (灰の道・鍛えの間・火守り。13 の 4 章) の部屋を撮る。./gradlew runShots -Pscript=tools/shots/ember_trials.txt
# 出力: run/shots/ember_trials/shots/ember_trials/*.png と shots.log。体は建てず、部屋の中身だけを余燼の背の箱に建てる (/skyabyss embertrial build)
# 向き: yaw 0 = 南 (z の正)、180 = 北。西の部屋 x -37..-13、東の部屋 x 13..39、南の部屋 x -10..10
world ember_trials flat
size 1280 720
timeout 600
render 12
clouds off

cmd gamemode spectator
cmd skyabyss embertrial build
cmd skyabyss embertrial forge 55
cmd skyabyss embertrial hearth 62
hud off

# 西 灰の道: 入口の岸から湖
camera skyabyss:ember_back -25 79.5 -6 0 12
waitchunks 900
wait 60
shot west_entrance
# 湖を高いところから (島 7 つ)
camera skyabyss:ember_back -25 92 -2 0 52
wait 20
shot west_lake_top
# 冷え殻の三つの段階 (黒・割れ目・沈みかけ) を島のそばに並べる
cmd execute in skyabyss:ember_back run fill -22 76 4 -20 76 6 skyabyss:cooled_crust[stage=0]
cmd execute in skyabyss:ember_back run fill -19 76 4 -17 76 6 skyabyss:cooled_crust[stage=1]
cmd execute in skyabyss:ember_back run fill -16 76 4 -14 76 6 skyabyss:cooled_crust[stage=2]
camera skyabyss:ember_back -18 79.5 -2 0 25
wait 20
shot west_crust_stages

# 東 鍛えの間: 大炉と鞴
camera skyabyss:ember_back 26 84 12 0 12
wait 40
shot east_forge
# 金床と刻み (北向き)
camera skyabyss:ember_back 26 83.5 17 180 20
wait 20
shot east_anvil_marks
# 大炉の色 (白 → 黒)
cmd skyabyss embertrial forge 95
camera skyabyss:ember_back 26 83.5 17 0 8
wait 20
shot east_forge_white
cmd skyabyss embertrial forge 0
wait 20
shot east_forge_cold

# 南 火守り: 谷の奥の火床と噴気
camera skyabyss:ember_back 0 82 36 0 8
wait 40
shot south_valley
cmd execute in skyabyss:ember_back run setblock 0 78 42 skyabyss:heat_vent[phase=1]
cmd execute in skyabyss:ember_back run setblock -4 78 46 skyabyss:heat_vent[phase=2]
cmd skyabyss embertrial hearth 100
camera skyabyss:ember_back 2 81 46 0 6
wait 20
shot south_hearth_vents
quit
